<?php
include 'db.php';

// সেশন চেক করা
if (!isset($_SESSION['role']) || $_SESSION['role'] !== 'parent') { 
    header("Location: login.php"); 
    exit(); 
}

$uid = $_SESSION['user_id'];

// ১. লগইন করা অভিভাবকের parent_id বের করা
$parent_res = $conn->query("SELECT parent_id FROM parents WHERE user_id = $uid");
$parent_data = $parent_res->fetch_assoc();
$parent_id = $parent_data['parent_id'] ?? 0;

// ২. অভিভাবকের সন্তানের প্রোফাইল ডাটা আনা
$stud_q = $conn->query("SELECT s.*, c.class_name FROM students s JOIN classes c ON s.class_id = c.class_id WHERE s.parent_id = $parent_id");
$student = $stud_q->fetch_assoc();
$student_id = $student['student_id'] ?? 0;

$report_card = [];

if($student_id > 0) {
    // সন্তানের ক্লাসের সব সাবজেক্টের লিস্ট আনা
    $subjects_q = $conn->query("SELECT cs.class_subject_id, sub.subject_name, ta.assignment_id 
                                FROM class_subjects cs 
                                JOIN subjects sub ON cs.subject_id = sub.subject_id
                                JOIN teacher_assignment ta ON cs.class_subject_id = ta.class_subject_id
                                WHERE cs.class_id = {$student['class_id']}");
    
    while($sub = $subjects_q->fetch_assoc()) {
        $asg_id = $sub['assignment_id'];
        
        // ক) সব কাস্টম অ্যাসেসমেন্টের প্রাপ্ত নম্বরের সরাসরি যোগফল (Direct Sum)
        $assess_res = $conn->query("SELECT SUM(sm.marks_obtained) as obtained 
                                    FROM student_marks sm 
                                    JOIN exam_configurations ec ON sm.exam_config_id = ec.exam_config_id 
                                    WHERE sm.student_id = $student_id AND ec.assignment_id = $asg_id AND ec.exam_type = 'assessment'");
        $assess_data = $assess_res->fetch_assoc();
        $final_assessment_mark = $assess_data['obtained'] ?? 0;
        
        // খ) ফাইনাল পরীক্ষার মার্কস বের করা (৭০ এর মধ্যে)
        $final_res = $conn->query("SELECT sm.marks_obtained as fnm FROM student_marks sm 
                                   JOIN exam_configurations ec ON sm.exam_config_id = ec.exam_config_id 
                                   WHERE sm.student_id = $student_id AND ec.assignment_id = $asg_id AND ec.exam_type = 'final'");
        $final_data = $final_res->fetch_assoc();
        $final_mark = ($final_data !== null) ? $final_data['fnm'] : null;
        
        // গ) মোট নম্বর (অ্যাসেসমেন্টের যোগফল + ফাইনাল মার্ক)
        $total_mark = ($final_mark !== null) ? ($final_assessment_mark + $final_mark) : $final_assessment_mark;
        $total_mark = round($total_mark, 2);
        
        // ঘ) লেটার গ্রেড ক্যালকুলেশন
        $grade = "Incomplete";
        if($final_mark !== null) {
            if($total_mark >= 80) $grade = "A+";
            elseif($total_mark >= 70) $grade = "A";
            elseif($total_mark >= 60) $grade = "A-";
            elseif($total_mark >= 50) $grade = "B";
            elseif($total_mark >= 40) $grade = "C";
            else $grade = "F";
        }
        
        $report_card[] = [
            'subject' => $sub['subject_name'],
            'assessment' => $final_assessment_mark,
            'final' => ($final_mark !== null) ? $final_mark : 'Not Published',
            'total' => $total_mark,
            'grade' => $grade
        ];
    }
}

// ৩. উপস্থিতি বা অ্যাটেনডেন্স কাউন্ট
$att_data = $conn->query("SELECT COUNT(*) as total, SUM(CASE WHEN status='Present' THEN 1 ELSE 0 END) as present FROM attendance WHERE student_id = $student_id")->fetch_assoc();
$total_days = $att_data['total'] ?? 0; 
$present_days = $att_data['present'] ?? 0;
$att_percentage = $total_days > 0 ? round(($present_days / $total_days) * 100, 2) : 0;
?>

$info_q = "SELECT c.class_id, c.class_name, s.subject_name FROM teacher_assignment ta 
           JOIN class_subjects cs ON ta.class_subject_id = cs.class_subject_id
           JOIN classes c ON cs.class_id = c.class_id 
           JOIN subjects s ON cs.subject_id = s.subject_id 
           WHERE ta.assignment_id = ?";
$stmt_info = $conn->prepare($info_q); 
$stmt_info->bind_param("i", $assignment_id); 
$stmt_info->execute();
$info_res = $stmt_info->get_result()->fetch_assoc();

if (!$info_res) { 
    die("Invalid Access or Assignment Not Found!");

$subject_name = $info_res['subject_name'];

$total_allocated_res = $conn->query("SELECT SUM(total_marks) as total_allocated FROM exam_configurations WHERE assignment_id = $assignment_id AND exam_type = 'assessment'");
$total_allocated_row = $total_allocated_res->fetch_assoc();
$current_assessment_total = $total_allocated_row['total_allocated'] ?? 0;
$remaining_marks = 30 - $current_assessment_total; // সর্বোচ্চ ৩০ থেকে বাকি কত আছে

// ১. দৈনিক উপস্থিতি (Attendance) সেভ করার লজিক

    $attendance_date = $_POST['attendance_date'];
    $present_students = $_POST['attendance'] ?? [];

    while ($st = $all_stud_res->fetch_assoc()) {
        $sid = $st['student_id'];
        $status = in_array($sid, $present_students) ? 'Present' : 'Absent';
        $conn->query("INSERT INTO attendance (student_id, assignment_id, attendance_date, status) VALUES ($sid, $assignment_id, '$attendance_date', '$status') ON DUPLICATE KEY UPDATE status = '$status'");

    $exam_title = trim($_POST['exam_title']);

        $msg = "Error! Exam marks must be greater than 0.";
        $msg_type = "danger";

    else if (($current_assessment_total + $input_marks) > 30) {

                header("Location: teacher_action.php?assignment_id=$assignment_id&active_exam=" . $conn->insert_id . "&msg=created");
